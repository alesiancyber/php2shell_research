<details markdown="block"><summary>Query</summary>

```kusto
let fatals =
    PhpErrorLog_CL
    | where Level == "Fatal error"
    | project FatalTime = TimeGenerated, Computer, Fatal = Message, Run = tostring(split(_ResourceId, "/")[1]);
let exits =
    SecurityEvent
    | where EventID == 4689 and ProcessName has "php-cgi"
    | project ExitTime = TimeGenerated, Computer;
fatals
| join kind=leftouter exits on Computer
| summarize MinGap = min(abs(datetime_diff('second', FatalTime, ExitTime))) by FatalTime, Computer, Fatal, Run
| extend Worker = iff(isnull(MinGap) or MinGap > 5, "REUSED (no restart)", "restarted")
| project FatalTime, Run, Computer, Fatal, Worker
| order by FatalTime asc
```
